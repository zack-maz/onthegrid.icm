    Add the module-private constants + `buildDeadUrlSample` helper (above the route handler):

    ```ts
    const LIMIT_DRILL_DOWN = 20;
    const MAX_SCAN_KEYS = 200;

    type DeadUrlSampleEntry = {
      eventId: string;
      url: string;
      status: 'dead-host' | '403' | '404';
    };

    async function buildDeadUrlSample(): Promise<DeadUrlSampleEntry[]> {
      try {
        const sample: DeadUrlSampleEntry[] = [];
        let cursor: number | string = 0;
        let scanned = 0;
        do {
          const [next, keys] = (await redis.scan(cursor, {
            match: `${URL_LIVENESS_KEY_PREFIX}*`,
            count: 50,
          })) as [string | number, string[]];
          cursor = next;
          for (const key of keys) {
            if (scanned >= MAX_SCAN_KEYS) { cursor = 0; break; }
            scanned += 1;
            const value = await cacheGetSafe<UrlLivenessValue>(key);
            if (!value) continue;
            if (!isTerminalDead(value.status)) continue;
            sample.push({
              eventId: key.replace(URL_LIVENESS_KEY_PREFIX, ''),
              url: value.lastUrlProbed,
              status: value.status as DeadUrlSampleEntry['status'],
            });
            if (sample.length >= LIMIT_DRILL_DOWN) { cursor = 0; break; }
          }
        } while (Number(cursor) !== 0);
        return sample;
      } catch (err) {
        log.warn({ err }, 'failed to build dead-URL sample');
        return [];
      }
    }
    ```
