    ```typescript
        // so the resolver's POI-amenity-disambiguation branch sees the same context

            {
              schemaVersion: 'v3',

              bellingcatByGroup: accumulatedBellingcat,
            },
            prioritizedGroups,
          );

            adapted = enrichedV3ToEntities(geo.events, prioritizedGroups);
          }
        } else if (pipelineV2) {
          const geo = await geocodeEnrichedEvents(
            {
              schemaVersion: 'v2',
              events: window as never,
              matchedNewsByGroup: accumulatedMatchedNews,
              bellingcatByGroup: accumulatedBellingcat,
            },
            prioritizedGroups,
          );

            adapted = enrichedV2ToEntities(geo.events, prioritizedGroups);
          }

          const geo = await geocodeEnrichedEvents(
            { schemaVersion: 'v1', events: window as never },
            prioritizedGroups,
          );
          if (geo.schemaVersion === 'v1') {

          }
        }

        if (adapted.length > 0) {

          lastFlushedEventCount = partialEvents.length;
          log.info(
            { completed, total, windowSize: window.length, flushed: adapted.length },
            'Plan 01 periodic-flush: incremental terminal-key write',
          );
        }
      } catch (flushErr) {
        // Best-effort — periodic flush failure must NOT abort the extraction loop.
        // The final flush at end-of-pipeline will still attempt the full-cache write.
        log.warn({ err: flushErr, completed, total }, 'Plan 01 periodic-flush failed (continuing)');
      }
    });
    ```

    **Step 3 — Surface per-batch matchedNews / bellingcat deltas to the callback.**

    The v3 extractor today populates `matchedNewsByGroup` / `bellingcatByGroup` on the run-result object only when processEventGroups resolves. To make periodic flushes use the SAME context the final flush uses, the pipeline needs to see those maps growing AS batches complete.

    **Chosen approach (closure-state via run-result reference):** Instead of widening the callback signature, the pipeline pre-allocates the two accumulator maps (Step 1) and passes them BY REFERENCE into `processEventGroups` via a new optional parameter. The v3 extractor then `set()`s entries into those maps inside its per-batch finishBatch chain (synchronously, before `await onBatchComplete?.()` runs). When the awaited callback fires, the maps already contain the just-completed batch's deltas.

    Concretely: widen `processEventGroups` (in the barrel `server/lib/llmEventExtractor.ts`) and the v3 extractor signature to accept an optional `accumulators?: { matchedNews: Map<string, NewsArticle[]>; bellingcat: Map<string, BellingcatGeoResult> }` parameter. The v3 extractor's per-batch finishBatch body sets entries into those maps before calling `await onBatchComplete?.()`. Existing callers that pass nothing get current behavior (extractor allocates its own maps internally).

    Pipeline call site:
    ```typescript
    const extractResult = await processEventGroups(
      prioritizedGroups,
      async (completed, total) => { /* ... callback body from Step 2 ... */ },
      { matchedNews: accumulatedMatchedNews, bellingcat: accumulatedBellingcat },
    );
    ```

    The race-safety contract (Phase 27.4.4 Plan 02): the v3 extractor's per-batch `finishBatch` does `accumulators.matchedNews.set(...)` SYNCHRONOUSLY between the batch's extraction promise resolution and the awaited `onBatchComplete?.()` call. Map.set is synchronous; the JS event loop is single-threaded; nothing else runs between the push and the callback. By the time the callback executes its periodic-flush body, the maps are byte-correct for the just-completed window.
