import { ME_VIEWBOX, ME_COUNTRY_CODES } from '../my_world/server/lib/meBounds.js';

async function check(landmark: string, amenity: string) {
  const r = await forwardGeocodeConstrained(landmark, {
    amenity,
    countrycodes: ME_COUNTRY_CODES,
    viewbox: ME_VIEWBOX,
    addressdetails: true,
    limit: 5,
  });

      null,
      2,
    ),
  );
  await new Promise((r) => setTimeout(r, 1100)); // 1 req/s throttle
}
