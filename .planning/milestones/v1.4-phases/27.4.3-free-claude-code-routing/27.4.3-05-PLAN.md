    **(C) Add D-17 Trigger 2 (eval-drop) — observation hook only**:

    The eval-drop trigger fires from a SCHEDULED or ON-DEMAND runEval call, not from inside the extractor batch loop. Add a helper export the harness can call AFTER `runEval()` finishes:

    ```typescript
    /**
     * D-17 Trigger 2: eval-drop auto-rollback. Called by llmEvalHarness after runEval
     * persists a new score; compares to the cutover-gate baseline at events:llm-eval-baseline:v3
     * (or the model-suffixed key for the active winner).
     *
     * If newScore drops ≥ 5pp from baseline AND v3 is active, flips override to v2.
     */
    export async function checkEvalDropTrigger(opts: { newWithin20kmRatio: number }): Promise<{ rolledBack: boolean; reason?: string }> {
      if (!isPipelineV3()) return { rolledBack: false };
      try {
        const baseline = await redis.get<{ within20km: number; total: number } | string | null>('events:llm-eval-baseline:v3');
        const baselineObj = typeof baseline === 'string' ? JSON.parse(baseline) : baseline;
        if (!baselineObj || typeof baselineObj.within20km !== 'number' || typeof baselineObj.total !== 'number' || baselineObj.total === 0) {
          return { rolledBack: false };
        }
        const baselineRatio = baselineObj.within20km / baselineObj.total;
        const drop = baselineRatio - opts.newWithin20kmRatio;
        if (drop >= 0.05) {
          log.warn({ drop, baselineRatio, newRatio: opts.newWithin20kmRatio }, 'D-17 Trigger 2: eval drop ≥ 5pp — auto-rollback v3 -> v2');
          setPipelineOverride('v2');
          try {
            await redis.set('events:llm-pipeline-override', 'v2', { ex: 7 * 24 * 3600 });
          } catch {}
          await appendPipelineAudit({
            ts: Date.now(),
            from: 'v3',
            to: 'v2',
            trigger: 'auto:eval_drop',
            operator: process.env.NODE_ENV === 'production' ? 'production' : 'cron',
            reason: `eval drop ${drop.toFixed(3)} (baseline ${baselineRatio.toFixed(3)} -> ${opts.newWithin20kmRatio.toFixed(3)})`,
          });
          return { rolledBack: true, reason: `eval drop ${drop.toFixed(3)}` };
        }
      } catch (err) {
        log.warn({ err }, 'eval-drop trigger check failed (redis unreachable?)');
      }
      return { rolledBack: false };
    }
    ```
