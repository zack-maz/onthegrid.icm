    /**
     * Phase 28.1 W6 D-13 — CSS-var → deck.gl RGBA tuple bridge.
     *
     * Reads --color-* CSS custom properties from :root once at module load and
     * parses to RGBA tuples for deck.gl getColor callbacks. Falls back to TS-literal
     * defaults when the CSS var is missing (SSR / jsdom / theme-not-loaded).
     *
     * Why module-load: deck.gl getColor callbacks fire per-frame; getComputedStyle
     * inside a callback is too expensive. The W6 D-13 contract is: CSS @theme is
     * source of truth for HTML/Tailwind, this module mirrors-once for deck.gl.
     *
     * Tradeoff: theme switching at runtime won't update colors without a reload.
     * Acceptable for 28.1 — there's only one theme.
     */

    function readCssRGB(varName: string, fallback: [number, number, number]): [number, number, number] {
      if (typeof window === 'undefined') return fallback; // SSR / jsdom
      const raw = getComputedStyle(document.documentElement).getPropertyValue(varName).trim();
      if (!raw) return fallback;
      if (raw.startsWith('#')) {
        const hex = raw.slice(1);
        return [
          parseInt(hex.slice(0, 2), 16),
          parseInt(hex.slice(2, 4), 16),
          parseInt(hex.slice(4, 6), 16),
        ];
      }
      const m = raw.match(/rgb\((\d+),\s*(\d+),\s*(\d+)\)/);
      if (m) return [Number(m[1]), Number(m[2]), Number(m[3])];
      return fallback;
    }

    function readCssHex(varName: string, fallback: string): string {
      if (typeof window === 'undefined') return fallback;
      const raw = getComputedStyle(document.documentElement).getPropertyValue(varName).trim();
      return raw || fallback;
    }
