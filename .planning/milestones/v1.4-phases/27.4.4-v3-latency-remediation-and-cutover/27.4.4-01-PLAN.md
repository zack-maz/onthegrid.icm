/**

/**
 * Phase 27.4.4 D-21 — fire-and-forget NIM pre-warm if last NIM call > 5min ago.
 *
 * Removes cold-start lag from p99. Smart-only: zero cost when warm (immediate
 * return); cheap (~1 call) when needed. Cost-shadow tracked in routingTrace.
 *
 * Called from processEventGroupsV3 BEFORE the main batch loop so cold starts
 * (Vercel scale-out, idle > 10min) trigger a 1-token warmup before the heavy
 * extraction batches. Failures are non-fatal — pre-warm is opportunistic.
 */
export async function prewarmIfCold(): Promise<void> {
  if (Date.now() - lastNimCallTs <= PREWARM_THRESHOLD_MS) return;
  const client = getNvidiaNimClient();
  if (!client) return;
  try {
    const t0 = Date.now();
    await client.chat.completions.create({
      model: NVIDIA_NIM_DEFAULT_MODEL,
      messages: [{ role: 'system', content: '.' }],
      max_tokens: 1,
      temperature: 0,
    });
    lastNimCallTs = Date.now();
    const prevTrace = llmProgress.routingTrace ?? [];
    updateProgress({
      routingTrace: [
        {
          ts: lastNimCallTs,
          batch: PREWARM_BATCH_SENTINEL,
          provider: 'nvidia_nim' as const,
          model: NVIDIA_NIM_DEFAULT_MODEL,
          reason: 'prewarm:cold-start',
        },
        ...prevTrace,
      ].slice(0, 50),
      prewarmCount: (llmProgress.prewarmCount ?? 0) + 1,
      lastPrewarmTs: lastNimCallTs,
      prewarmState: 'cold-fired' as const,
    });
    logger
      .child({ module: 'pre-warm' })
      .info({ durationMs: Date.now() - t0 }, 'NIM pre-warm fired (cold-start)');
  } catch (err) {
    logger
      .child({ module: 'pre-warm' })
      .warn(
        { err: err instanceof Error ? err.message : String(err) },
        'NIM pre-warm failed (non-fatal)',
      );
  }
}

import { redis } from '../server/cache/redis.js';

async function main(): Promise<void> {
  const labelArg = process.argv.find((a) => a.startsWith('--label='));
