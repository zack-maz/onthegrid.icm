    ```
    import { canonicalize } from '../data/actor-catalog.js';

    /** Phase 33 D-08 — walk each event's actors[] through the canonical catalog. */
    /**  Matched aliases replaced by canonicalName; unmatched pass through unchanged. */
    /**  Catalog is the single source of truth; LLM prompt compliance is best-effort (D-09). */
    export function applyCatalogToEvents(events: EnrichedEventV3[]): EnrichedEventV3[] {
      return events.map((event) => ({
        ...event,
        actors: event.actors.map((a) => canonicalize(a)?.canonicalName ?? a),
      }));
    }

    /** Phase 33 D-10 — server-side defense-in-depth for actorConfidence. */
    /**  Fills missing entries OR length-mismatched arrays with 'low' defaults so the */
    /**  index-locked-parallel-array invariant (D-10) holds for every cache write. */
    /**  Conservative 'low' default means day-1 dashboard counts are inflated until */
    /**  the LLM begins emitting actorConfidence reliably (Open Q §2 monitoring window). */
    export function repairActorConfidence(event: EnrichedEventV3): EnrichedEventV3 {
      if (
        event.actorConfidence == null ||
        event.actorConfidence.length !== event.actors.length
      ) {
        return {
          ...event,
          actorConfidence: event.actors.map(() => 'low' as const),
        };
      }
      return event;
    }
    ```
