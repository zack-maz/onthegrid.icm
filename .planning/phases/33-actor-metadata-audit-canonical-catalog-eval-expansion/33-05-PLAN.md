    ```
    // === PHASE 33 D-13 — actor-match scoring over live cache ===
    // Open Q §4 resolution (path c): match by landmark + country case-insensitive
    // substring because ground-truth id is synthetic gt-NNN, NOT a real groupKey.
    // Direct groupKey === id join (as D-13 originally specified) finds zero matches.

    let actorMatched = 0;
    let actorTotal = 0;

    try {
      const live = await cacheGetSafe<ConflictEventEntity[]>(LLM_EVENTS_KEY_ACTIVE, 999_999_999);
      const liveEvents: ConflictEventEntity[] = live?.data ?? [];

      for (const gtEvent of gt.events) {
        const exp1 = (gtEvent as { expectedActor1?: string | null }).expectedActor1;
        const exp2 = (gtEvent as { expectedActor2?: string | null }).expectedActor2;
        if (exp1 == null || typeof exp1 !== 'string') continue;
        actorTotal++;
        const landmark = (gtEvent.hierarchy.landmark ?? '').toLowerCase();
        const country = (gtEvent.hierarchy.country ?? '').toLowerCase();
        const exp1Low = exp1.toLowerCase();
        const exp2Low = typeof exp2 === 'string' ? exp2.toLowerCase() : null;

        const candidates = liveEvents.filter((e) => {
          const label = (e.label ?? '').toLowerCase();
          return (landmark.length > 0 && label.includes(landmark))
              || (country.length > 0 && label.includes(country));
        });
        for (const cand of candidates) {
          const actors = ((cand.data as { actors?: string[] }).actors ?? []).map((a) => a.toLowerCase());
          const has1 = actors.some((a) => a.includes(exp1Low));
          const has2 = exp2Low == null || actors.some((a) => a.includes(exp2Low));
          if (has1 && has2) {
            actorMatched++;
            break; // count once per gt event
          }
        }
      }
    } catch (err) {
      log.warn({ err }, 'failed to compute actorMatchRate; scoring will record 0');
    }

    const actorMatchRate = actorTotal === 0 ? 0 : actorMatched / actorTotal;
    ```
