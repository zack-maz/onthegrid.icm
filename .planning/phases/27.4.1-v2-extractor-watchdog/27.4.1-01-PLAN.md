```typescript
function withTimeout<T>(promise: Promise<T>, ms: number, label: string): Promise<T> {
  let timer: ReturnType<typeof setTimeout> | undefined;
  const timeout = new Promise<never>((_, reject) => {
    timer = setTimeout(() => reject(new Error(`${label} timed out after ${ms}ms`)), ms);
  });
  return Promise.race([promise, timeout]).finally(() => {
    if (timer) clearTimeout(timer);
  });
}
```

Watchdog contract (authoritative shape — implement exactly this):

```typescript
export interface BatchWatchdogOptions {
  timeoutMs: number; // hard kill threshold (D-01 default 90_000)
  softWarnMs: number; // soft warn log (D-02 default 60_000)
  batchIndex: number; // for error/log correlation
  label: string; // 'v1' | 'v2' — ends up in log fields + timeout msg
  onTimeout: () => Promise<void>; // caller composes DLQ enqueue + progress increment
  onSoftWarn?: (elapsedMs: number) => void; // called once if soft threshold crosses mid-call
}

/**
 * Wrap a batch promise with timeout + soft-warn.
 * Returns T on success, null on timeout (late-resolving promise is discarded).
 * Throws only on programmer errors; batchFn rejections propagate unchanged.
 */
export async function withBatchWatchdog<T>(
  batchFn: () => Promise<T>,
  opts: BatchWatchdogOptions,
): Promise<T | null>;
```
