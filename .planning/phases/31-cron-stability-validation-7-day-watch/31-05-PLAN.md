    const log = JSON.parse(require('fs').readFileSync('.planning/phases/31-cron-stability-validation-7-day-watch/watch-log.json','utf8'));
    const natural = log.rows.filter(r => r.natural === true);
    const pass = natural.filter(r => r.result === 'PASS');
    const fail = natural.filter(r => r.result === 'FAIL');
    const gap = natural.filter(r => r.result === 'GAP');
    const dlqCounts = pass.map(r => r.dlq.count);
    const evalAt20km = pass.map(r => r.eval.at20km).filter(x => x !== null);
    const batchCounts = pass.map(r => r.batchCount);
    const breakerTrips = pass.reduce((sum, r) => sum + r.breakerTrips, 0);
    const stats = {
      totalRows: log.rows.length,
      naturalRows: natural.length,
      passDays: pass.length,
      failDays: fail.length,
      gapDays: gap.length,
      dlqCountMin: Math.min(...dlqCounts),
      dlqCountMax: Math.max(...dlqCounts),
      dlqCountMean: (dlqCounts.reduce((a,b)=>a+b,0) / dlqCounts.length).toFixed(2),
      evalAt20kmMean: evalAt20km.length ? (evalAt20km.reduce((a,b)=>a+b,0) / evalAt20km.length).toFixed(3) : 'n/a',
      batchCountMin: Math.min(...batchCounts),
      batchCountMax: Math.max(...batchCounts),
      batchCountMean: (batchCounts.reduce((a,b)=>a+b,0) / batchCounts.length).toFixed(0),
      breakerTripsTotal: breakerTrips,
      firstTickDate: log.rows[0].tickDate,
      lastTickDate: log.rows[log.rows.length - 1].tickDate,
    };
    console.log(JSON.stringify(stats, null, 2));
    "
