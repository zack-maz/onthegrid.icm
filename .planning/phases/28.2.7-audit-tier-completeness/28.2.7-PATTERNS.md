```typescript
async function probeCronTick(name: string): Promise<ProbeResult> {
  const start = Date.now();
  try {
    const entry = await withTimeout(
      cacheGetSafe<number>(`cron:lastTick:${name}`, 999_999_999),
      PROBE_TIMEOUT_MS,
      `probe cron ${name}`,
    );
    const latencyMs = Date.now() - start;
    if (entry === null) {
      return {
        freshnessMs: null,
        lastSuccessTs: null,
        hadError: false,
        errorReason: null,
        latencyMs,
      };
    }
    const tickTs = typeof entry.data === 'number' ? entry.data : entry.lastFresh;
    return {
      freshnessMs: Date.now() - tickTs,
      lastSuccessTs: tickTs,
      hadError: false,
      errorReason: null,
      latencyMs,
    };
  } catch (err) {
    return {
      freshnessMs: null,
      lastSuccessTs: null,
      hadError: true,
      errorReason: sanitizeError(err),
      latencyMs: Date.now() - start,
    };
  }
}
```

**Pattern to apply** — refactor `probeLlmStatus` to async, Redis-first with in-memory fallback (CONTEXT D-08 + D-09 + D-10):

```typescript
/**
 * Probe `/api/llm-status` via Redis-backed `llm:lastProgress` key with
 * in-memory `llmProgress` singleton fallback (Phase 28.2.7 R2 D-08).
 *
 * Read order: Redis first → in-memory fallback. Solves the cold-start
 * problem where a fresh Vercel function instance hasn't run the cron and
 * sees the singleton at INITIAL_PROGRESS.
 *
 * D-09: when BOTH null, return freshnessMs:null → 'unknown'. First cron
 * tick within 26h flips green. No artificial "fresh deploy → healthy" bias.
 *
 * D-10: read-only — no backfill of singleton from Redis. Singleton becomes
 * eventually consistent via the next pipeline's resetProgress call.
 */
async function probeLlmStatus(): Promise<ProbeResult> {
  const start = Date.now();
  // Redis read — degrade-open via cacheGetSafe (in-memory fallback already
  // built into the helper; throws bubble to the catch below for the
  // singleton fallback).
  let redisLatest: number | null = null;
  try {
    const entry = await withTimeout(
      cacheGetSafe<{ startedAt: number | null; completedAt: number | null }>(
        'llm:lastProgress',
        999_999_999,
      ),
      PROBE_TIMEOUT_MS,
      'probe llmStatus (redis)',
    );
    if (entry?.data) {
      redisLatest = entry.data.completedAt ?? entry.data.startedAt ?? null;
    }
  } catch {
    // Fall through to in-memory singleton.
  }
  const memLatest = llmProgress.completedAt ?? llmProgress.startedAt ?? null;
  const latest = redisLatest ?? memLatest;
  return {
    freshnessMs: latest === null ? null : Date.now() - latest,
    lastSuccessTs: latest,
    hadError: false,
    errorReason: null,
    latencyMs: Date.now() - start,
  };
}
```

- The shape `{ startedAt, completedAt }` matches the schema written by `llmProgress.ts` setters per CONTEXT D-01.

```typescript
function createReqRes(headers: Record<string, string> = {}, query: Record<string, string> = {}) {
  const lower: Record<string, string> = {};
  for (const [k, v] of Object.entries(headers)) lower[k.toLowerCase()] = v;
  const req = { headers: lower, header: (name: string) => lower[name.toLowerCase()], query }
    as unknown as import('express').Request;
  let statusCode = 200;
  let body: unknown;
  const res = {
    status(code: number) { statusCode = code; return res; },
    json(data: unknown) { body = data; return res; },
  } as unknown as import('express').Response;
  return { req, res, getStatus: () => statusCode, getBody: () => body as Record<string, unknown> };
}

interface RouteLayer {
  route?: { methods: Record<string, boolean>; stack: Array<{ handle: Function }> };
}
function extractHandler(router: ReturnType<typeof Router>) {
  const stack = (router as unknown as { stack: RouteLayer[] }).stack;
  for (const layer of stack) {
    if (layer.route?.methods.get) return layer.route.stack[0].handle;
  }
  throw new Error('No GET handler found');
}
```

### `cacheSetSafe` write — degrade-open contract

**Source:** `server/cache/redis.ts:234-242`

```typescript
export async function cacheSetSafe<T>(key: string, data: T, redisTtlSec: number): Promise<void> {
  // Always update memCache first
  memCache.set(key, { data, fetchedAt: Date.now() });
  try {
    await withTimeout(cacheSet(key, data, redisTtlSec), REDIS_OP_TIMEOUT_MS, `cacheSet(${key})`);
  } catch {
    // Swallow Redis error or timeout -- memCache is already updated
  }
}
```

**Apply to:** All R1 + R2 Redis writes. CONTEXT D-11 confirms: pass `Date.now()` directly; envelope `{data, fetchedAt}` is added by `cacheSet` internally; `probeCronTick` already handles both `entry.data` and `entry.lastFresh` shapes.

**Key invariants:**

- ✓ Never throws (callers don't need try/catch)
- ✓ memCache populated regardless of Redis success (so the same instance reads its own writes immediately)
- ✓ TTL is seconds (not ms) — pass `CRON_LASTTICK_TTL_SEC = 604_800` directly

**Source:** `server/cache/redis.ts:126-129`
