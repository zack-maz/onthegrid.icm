```javascript
// ----- Phase 28.2.5 D-09 — Tier-green assertion (Band B gate) -----
// Fetch /api/health and assert the D-03 tier map. Workflow exits non-zero
// if any tier is red, so the GH Actions run signals the gate state cleanly.
// Bearer-bypassed via Phase 28.2 D-04 — /api/health has no per-endpoint
// limiter, the global tier accepts the Bearer.
let allTiersGreen = false;
const tierStatus = {
  critical: 'unknown',
  nonCritical: 'unknown',
  static: 'unknown',
  probeOnly: 'unknown',
  cron: 'unknown',
};
try {
  const healthRes = await fetch(`${process.env.API_BASE_URL}/api/health`, {
    headers: { Authorization: `Bearer ${process.env.DASHBOARD_PASSWORD}` },
  });
  if (!healthRes.ok) {
    throw new Error(`/api/health returned ${healthRes.status}`);
  }
  const health = await healthRes.json();
  // Map healthSources.ts TIER_BY_ENDPOINT slugs (kebab-case) to the
  // tierStatus camelCase keys used in the AuditPayload contract.
  const TIER_TO_KEY = {
    critical: 'critical',
    'non-critical': 'nonCritical',
    static: 'static',
    'probe-only': 'probeOnly',
    cron: 'cron',
  };
  // Per-tier rollup: status is "healthy" iff ALL endpoints in the tier
  // are healthy; "degraded" if any are degraded but none unhealthy;
  // "unhealthy" if any are unhealthy.
  const buckets = {};
  for (const ep of Object.values(health.endpoints || {})) {
    const tierKey = TIER_TO_KEY[ep.tier];
    if (!tierKey) continue;
    buckets[tierKey] = buckets[tierKey] || [];
    buckets[tierKey].push(ep.status);
  }
  for (const [tierKey, statuses] of Object.entries(buckets)) {
    if (statuses.some((s) => s === 'unhealthy')) tierStatus[tierKey] = 'unhealthy';
    else if (statuses.some((s) => s === 'unknown')) tierStatus[tierKey] = 'unknown';
    else if (statuses.some((s) => s === 'degraded')) tierStatus[tierKey] = 'degraded';
    else tierStatus[tierKey] = 'healthy';
  }
  // D-03 truth table:
  //   critical    → MUST be "healthy"
  //   non-critical → MAY be "healthy" or "degraded"
  //   static      → MAY be "healthy" or "degraded"
  //   probe-only  → MUST be "healthy"
  //   cron        → MAY be "healthy" or "degraded" (≤26h covers daily skew)
  const okNonCritical = ['healthy', 'degraded'].includes(tierStatus.nonCritical);
  const okStatic = ['healthy', 'degraded'].includes(tierStatus.static);
  const okCron = ['healthy', 'degraded'].includes(tierStatus.cron);
  allTiersGreen =
    tierStatus.critical === 'healthy' &&
    okNonCritical &&
    okStatic &&
    tierStatus.probeOnly === 'healthy' &&
    okCron;
  console.log('Tier-green assertion:', JSON.stringify({ allTiersGreen, tierStatus }, null, 2));
} catch (err) {
  console.error('Tier-green assertion failed to fetch /api/health:', err.message);
  // allTiersGreen stays false; tierStatus stays "unknown"
}
```
