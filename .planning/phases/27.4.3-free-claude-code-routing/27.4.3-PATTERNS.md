type RouterErrorBucket =
  | 'rate_limit'
  | 'timeout'
  | 'malformed_json'
  | 'schema_fail'
  | 'network'
  | 'upstream_500'
  | 'other';

function classifyError(err: unknown): RouterErrorBucket {
  if (err instanceof Error) {
    const m = err.message.toLowerCase();
    if (m.includes('429') || m.includes('rate limit')) return 'rate_limit';
    if (m.includes('timeout') || m.includes('timed out')) return 'timeout';
    if (m.includes('enotfound') || m.includes('econnreset') || m.includes('eai_again'))

    if (m.match(/\b5\d\d\b/)) return 'upstream_500';
  }
  return 'other';
}

## Shared Patterns

### Defense-in-depth `coerceCachedEvents` at every cache read site

**Source:** `server/routes/events.ts:447-454`
**Apply to:** every new v3 cache read in `events.ts` (cache-fallback chain, /llm-replay, /llm-status's `loadRecentEnrichedEvents`)

```typescript
function toEntityArray(data: unknown): ConflictEventEntity[] {
  return Array.isArray(data) ? (data as ConflictEventEntity[]) : [];
}

function coerceCachedEvents<C extends { data: unknown }>(
  cached: C,
): Omit<C, 'data'> & { data: ConflictEventEntity[] } {
  return { ...cached, data: toEntityArray(cached.data) };
}
```

**Why:** 27.4.1 commit `e26ceca` post-mortem — partial-cache envelope writes to terminal key crashed `events.map` consumers. Coercion at every read site means downstream iterations and array methods see a guaranteed `ConflictEventEntity[]` even if a future regression reintroduces envelope-shaped cache data.

### Pipeline-flip audit log (D-15 — NEW shared primitive)

**Source:** no existing analog — pattern derived from `llmDLQ.ts:39-69` (bounded-set + TTL pattern) and `events:llm-pipeline-override` Redis key (existing override pattern at `server/routes/events.ts:473-492`).

**Apply to:** every place that calls `setPipelineOverride()` — currently `events.ts /llm-pipeline POST` handler, future v3 auto-rollback hooks (D-17 trigger 1 + 2).

```typescript
// New helper module OR inlined in server/routes/events.ts:
const PIPELINE_AUDIT_KEY = 'events:llm-pipeline-audit';
const PIPELINE_AUDIT_TTL_SEC = 90 * 24 * 3600;
const PIPELINE_AUDIT_MAX = 200;

interface PipelineFlipEntry {
  ts: number;
  from: 'v1' | 'v2' | 'v3';
  to: 'v1' | 'v2' | 'v3';
  trigger: 'manual:operator_post' | 'auto:eval_drop' | 'auto:watchdog_recurrence';
  operator: 'dev' | 'production' | 'cron';
  reason?: string;
}

async function appendPipelineAudit(entry: PipelineFlipEntry): Promise<void> {
  try {
    await redis.lpush(PIPELINE_AUDIT_KEY, JSON.stringify(entry));
    await redis.ltrim(PIPELINE_AUDIT_KEY, 0, PIPELINE_AUDIT_MAX - 1);
    await redis.expire(PIPELINE_AUDIT_KEY, PIPELINE_AUDIT_TTL_SEC);
  } catch (err) {
    log.warn({ err, entry }, 'audit append failed (redis unreachable)');
  }
}

async function listPipelineAudit(limit: number): Promise<PipelineFlipEntry[]> {
  try {
    const raw = await redis.lrange(PIPELINE_AUDIT_KEY, 0, limit - 1);
    return raw
      .map((s) => JSON.parse(typeof s === 'string' ? s : JSON.stringify(s)))
      .filter(Boolean);
  } catch {
    return [];
  }
}
```

**Pattern source for LREVERSE-style bounded list:** `llmDLQ.ts:39-69` uses `sadd` + `srem` for a set; the audit log uses `lpush` + `ltrim` for an ordered list (newest-first). The graceful-degradation pattern (try/catch + warn-log) is identical.
