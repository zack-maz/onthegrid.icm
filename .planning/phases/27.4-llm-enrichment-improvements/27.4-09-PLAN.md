```typescript
// ... at top of file

eventsRouter.get('/llm-status', async (_req, res) => {
  if (process.env.NODE_ENV === 'production') return res.status(404).json({ error: 'Not found' });

  // Compose full v2 observability payload.
  const dlqRecent = await listDLQ(50);
  const common = {
    schemaVersion: llmProgress.schemaVersion,
    callHistory: llmProgress.callHistory,
    tokenCounters: llmProgress.tokenCounters,
    dlqCount: llmProgress.dlqCount ?? dlqRecent.length,
    dlqRecent,
    breakerState: llmProgress.breakerState,
    evalScore: llmProgress.evalScore,
    provenanceCounts: llmProgress.provenanceCounts,
    suspectCount: llmProgress.suspectCount,
  };

  if (llmProgress.stage !== 'idle') {
    return res.json({ ...llmProgress, ...common });
  }

  const summaryActive = await cacheGetSafe<LLMRunSummary>(LLM_SUMMARY_KEY_ACTIVE, 0);
  if (summaryActive?.data) {
    return res.json({ stage: 'idle' as const, lastRun: summaryActive.data, ...common });
  }
  res.json({ stage: 'idle' as const, lastRun: null, ...common });
});
```

HistogramsBlock (D-17) — provenance distribution + call success rate:

```tsx
function HistogramsBlock({
  provenanceCounts,
  callHistory,
}: {
  provenanceCounts: Record<string, number>;
  callHistory: NonNullable<LLMStatus['callHistory']>;
}) {
  const provEntries = Object.entries(provenanceCounts).sort((a, b) => b[1] - a[1]);
  const totalProv = provEntries.reduce((s, [, n]) => s + n, 0);
  const okCount = callHistory.filter((c) => c.ok).length;
  return (
    <div className="mt-2">
      <div className="text-[9px] font-bold uppercase text-white/40">Provenance Distribution</div>
      {provEntries.length === 0 ? (
        <div className="text-[9px] text-white/40">no data</div>
      ) : (
        provEntries.map(([key, n]) => (
          <div key={key} className="mt-0.5">
            <div className="flex items-center justify-between text-[9px]">
              <span className="text-white/60">{key}</span>
              <span className="text-white/40 tabular-nums">
                {n} / {totalProv}
              </span>
            </div>
            <ProgressBar completed={n} total={totalProv || 1} />
          </div>
        ))
      )}
      <div className="mt-2 text-[9px] text-white/60">
        LLM Call Success:{' '}
        <span className="text-white/80 tabular-nums">
          {okCount}/{callHistory.length}
        </span>
      </div>
    </div>
  );
}
```

```tsx
function DrillDownBlock({ llmStatus }: { llmStatus: LLMStatus }) {
  const [expanded, setExpanded] = React.useState(false);
  const [copyFeedback, setCopyFeedback] = React.useState<string | null>(null);
  async function copyPayload() {
    try {
      await navigator.clipboard.writeText(JSON.stringify(llmStatus, null, 2));
      setCopyFeedback('Copied!');

    }

  return (
    <div className="mt-2">
```
