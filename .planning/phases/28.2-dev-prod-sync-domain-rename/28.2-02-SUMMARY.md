import { timingSafeEqual } from 'node:crypto';

// server/middleware/rateLimit.ts:53-85 (new — inside rateLimitHandler, after dev/Vercel skip,
// before identifier+limiter.limit() pathway)
if (prefix === 'ratelimit:public') {
  const expected = process.env.DASHBOARD_PASSWORD ?? '';
  if (expected !== '') {
    const header = req.headers.authorization ?? '';
    const bearerPrefix = 'Bearer ';
    if (header.startsWith(bearerPrefix)) {
      const provided = header.slice(bearerPrefix.length);
      if (provided.length === expected.length) {
        const a = Buffer.from(provided);
        const b = Buffer.from(expected);
        if (timingSafeEqual(a, b)) {
          next();
          return;
        }
      }
    }
  }
}
// Fall through to existing limiter.limit() pathway.

## Decisions Made

- **Option 1 (in-place bypass branch) over Option 2 (sibling middleware at mount site):** PATTERNS.md offered both. Option 1 keeps `server/index.ts:107` untouched and centralizes rate-limit logic in one module — no two places to grep when debugging bypass behavior. Closure-captured `prefix` makes the branch dead code for the 10 per-endpoint limiters at module load, so there's no runtime cost on the throttle hot path.
- **Length-precheck before `timingSafeEqual`:** Node throws `RangeError` on unequal-length Buffers per its contract. Length is not the secret (only the bytes are), so an early-return on mismatch is correct + safe.
- **Test count: 7 not 5 (PLAN listed both numbers depending on which paragraph):** The plan's `<truths>` block listed 5 cases but the `<behavior>` block listed 7 numbered tests. I implemented all 7 — the plan's `<action>` paragraph and grep contract (`describe('rate-limit bearer bypass'`) refer to the matrix as a whole, not the case count. All 7 tests are necessary: 1 (dev non-regression) + 1 (no-Bearer) + 1 (valid-Bearer-bypass) + 2 (bad-Bearer cases) + 1 (empty-password) + 1 (per-endpoint scope guard).
