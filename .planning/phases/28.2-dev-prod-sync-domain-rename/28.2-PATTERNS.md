**Constant-time compare excerpt to copy from `dashboardAuth.ts:51-66`:**

```typescript
const expected = process.env.DASHBOARD_PASSWORD ?? '';
if (expected === '') {
  // No password configured → no bypass; fall through to existing limiter.
  // (NB: this differs from dashboardAuth which 503s — for rate-limit bypass,
  // an unconfigured password is "no privileged callers", not "outage".)
}
const header = req.headers.authorization ?? '';
const prefix = 'Bearer ';
if (header.startsWith(prefix) && expected !== '') {
  const provided = header.slice(prefix.length);
  if (provided.length === expected.length) {
    const a = Buffer.from(provided);
    const b = Buffer.from(expected);
    if (timingSafeEqual(a, b)) {
      next();
      return; // Bearer match → skip global throttle
    }
  }
}
// Fall through to existing limiter.limit() pathway
```

```typescript
// AI-SPEC §6 — per-Bearer replay quota: 50 calls / 24h rolling
import { createHash } from 'node:crypto';

const fingerprint = createHash('sha256')
  .update(process.env.DASHBOARD_PASSWORD ?? '')
  .digest('hex')
  .slice(0, 8);
const quotaKey = `operator:replay-quota:${fingerprint}:${new Date().toISOString().slice(0, 10)}`;
const count = await redis.incr(quotaKey);
if (count === 1) {
  await redis.expire(quotaKey, 48 * 3600); // 48h TTL on first set
}
if (count > 50) {
  // UI-SPEC §6.2 + AI-SPEC §6 — 429 with Retry-After
  res.set('Retry-After', String(/* seconds until 24h boundary */));
  return res.status(429).json({
    error: 'replay_quota_exceeded',
    message: 'Replay quota reached: 50 of 50 in last 24h.',
    resetsAt: /* ISO timestamp */,
  });
}
```

// AI-SPEC §6 — operator-action audit log: bounded 500 entries, 30d TTL
const auditEntry = JSON.stringify({
  timestamp: Date.now(),
  bearerFingerprint: fingerprint, // SHA-256 truncated, never the raw secret
  operation: 'pipeline-swap', // or 'replay'
  args: { version, groupKey }, // sanitized; no PII
  result: 'ok',
});
try {
  await redis.sadd('operator:audit-log', auditEntry);
  const card = await redis.scard('operator:audit-log');
  if (card > 500) {
    // Evict oldest by SPOP raw-string srem (matches llmDLQ.ts pattern)
  }
  await redis.expire('operator:audit-log', 30 * 86400);

  // Audit log degraded → continue; never blocks operator action
}
